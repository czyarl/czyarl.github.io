{{- $registry := hugo.Data.tags -}} {{- $aliases := collections.NewScratch -}} {{- range $canonical, $meta := $registry -}} {{- if eq (len (findRE `^[a-z0-9]+(-[a-z0-9]+)*$` $canonical)) 0 -}} {{- errorf "Canonical tag %q must use lowercase ASCII kebab-case." $canonical -}} {{- end -}} {{- if not $meta.label -}} {{- errorf "Canonical tag %q is missing a label in data/tags.yaml." $canonical -}} {{- end -}} {{- if not $meta.description -}} {{- errorf "Canonical tag %q is missing a description in data/tags.yaml." $canonical -}} {{- end -}} {{- range ($meta.aliases | default slice) -}} {{- $alias := printf "%v" . -}} {{- if collections.IsSet $registry $alias -}} {{- errorf "Tag alias %q for %q is also a canonical tag." $alias $canonical -}} {{- end -}} {{- with $aliases.Get $alias -}} {{- errorf "Tag alias %q is registered for both %q and %q." $alias . $canonical -}} {{- else -}} {{- $aliases.Set $alias $canonical -}} {{- end -}} {{- end -}} {{- end -}} {{- range where site.RegularPages "Section" "posts" -}} {{- $page := . -}} {{- $tags := .Params.tags | default slice -}} {{- if eq (len $tags) 0 -}} {{- errorf "Post %s must have at least one canonical tag." $page.File.Path -}} {{- end -}} {{- if gt (len $tags) 4 -}} {{- errorf "Post %s has %d tags; keep at most four meaningful tags." $page.File.Path (len $tags) -}} {{- end -}} {{- $seen := collections.NewScratch -}} {{- range $tags -}} {{- $tag := printf "%v" . -}} {{- if ne $tag (lower $tag) -}} {{- errorf "Tag %q in %s must use lowercase canonical spelling." $tag $page.File.Path -}} {{- end -}} {{- if not (collections.IsSet $registry $tag) -}} {{- errorf "Unknown tag %q in %s. Register it in data/tags.yaml before publishing." $tag $page.File.Path -}} {{- end -}} {{- if $seen.Get $tag -}} {{- errorf "Duplicate tag %q in %s." $tag $page.File.Path -}} {{- end -}} {{- $seen.Set $tag true -}} {{- end -}} {{- end -}}
